// 1. Box for Heap Allocation and Indirection
// Box allows recursive structures (like trees/nodes) by fixing the type size on the stack.

enum TreeNode {
    Leaf(i32),
    // Without Box, TreeNode would have an infinite size at compile time!
    Branch {
        left: Box,
        right: Box,
    },
}

fn calculate_sum(node: &TreeNode) -> i32 {
    match node {
        TreeNode::Leaf(val) => *val,
        TreeNode::Branch { left, right } => calculate_sum(left) + calculate_sum(right),
    }
}

fn main() {
    let tree = TreeNode::Branch {
        left: Box::new(TreeNode::Leaf(10)),
        right: Box::new(TreeNode::Leaf(20)),
    };

    println!("Total tree sum: {}", calculate_sum(&tree));
}
// 2. Rc>: Multiple Ownership + Interior Mutability (Single-Threaded)
use std::rc::Rc;
use std::cell::RefCell;

struct Logger {
    logs: Vec,
}

impl Logger {
    fn log(&mut self, msg: &str) {
        self.logs.push(msg.to_string());
    }
}

fn main() {
    // Shared reference-counted logger with interior mutability
    let shared_logger = Rc::new(RefCell::new(Logger { logs: Vec::new() }));

    // Component A holds a clone of the reference counter
    let client_a = Rc::clone(&shared_logger);
    client_a.borrow_mut().log("Client A connected");

    // Component B holds another clone
    let client_b = Rc::clone(&shared_logger);
    client_b.borrow_mut().log("Client B performed action");

    // Inspect logs safely
    println!("Log Count: {}", shared_logger.borrow().logs.len());
}
// 3. Arc>: Thread-Safe Atomic Reference Counting & Locking
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    // Atomic Reference Counter wrapped around a Mutex lock
    let counter = Arc::new(Mutex::new(0));
    let mut handles = vec![];

    for _ in 0..10 {
        let counter_clone = Arc::clone(&counter);
        let handle = thread::spawn(move || {
            // Lock guarantees exclusive mutable access across threads
            let mut num = counter_clone.lock().unwrap();
            *num += 1;
        });
        handles.push(handle);
    }

    for handle in handles {
        handle.join().unwrap();
    }

    println!("Final Counter Result: {}", *counter.lock().unwrap());
}

Level 4 Architectural Pillars

Box<T> Heap Indirection

Puts data on the heap while keeping a fixed-size pointer on the stack. Crucial for dynamic dispatch (Box<dyn Trait>) and recursive data structures.

CS Principle: Fixed Stack Pointer to Dynamic Memory

Rc<T> & Arc<T> Reference Counting

Enables multiple read-only owners. When the reference count reaches zero, memory cleanup triggers automatically via RAII (Drop).

CS Principle: Shared Immutable DAG Ownership

RefCell<T> Interior Mutability

Bypasses compile-time borrow rules by checking borrows dynamically at runtime. Allows mutating values even through immutable standard references.

CS Principle: Dynamic Borrow Checking

Deref & Drop Traits

Deref makes custom smart pointers act like standard references. Drop customizes cleanup operations when a value leaves its scope.

CS Principle: Deterministic Resource Management (RAII)
