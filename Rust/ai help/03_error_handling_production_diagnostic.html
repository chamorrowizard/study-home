// src/error.rs
//! Custom Domain Error Types
//! Implementing `Display` and `std::error::Error` for strongly-typed library errors.

use std::fmt;

#[derive(Debug)]
pub enum BankError {
    InsufficientFunds { requested: f64, available: f64 },
    AccountLocked(String),
    NetworkTimeout,
}

// 1. Human-readable formatting for logs/display
impl fmt::Display for BankError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            BankError::InsufficientFunds { requested, available } => {
                write!(f, "Cannot withdraw ${requested:.2}; balance is ${available:.2}")
            }
            BankError::AccountLocked(id) => write!(f, "Account '{id}' is locked for safety"),
            BankError::NetworkTimeout => write!(f, "Banking network request timed out"),
        }
    }
}

// 2. Trait integration allowing this error to integrate with `std::error::Error`
impl std::error::Error for BankError {}
// src/bank.rs
//! Business Logic utilizing the `?` operator for clean error propagation.

use crate::error::BankError;

pub struct Account {
    pub id: String,
    balance: f64,
    is_locked: bool,
}

impl Account {
    pub fn new(id: &str, initial: f64) -> Self {
        Self {
            id: id.to_string(),
            balance: initial,
            is_locked: false,
        }
    }

    pub fn withdraw(&mut self, amount: f64) -> Result {
        if self.is_locked {
            return Err(BankError::AccountLocked(self.id.clone()));
        }

        if amount > self.balance {
            return Err(BankError::InsufficientFunds {
                requested: amount,
                available: self.balance,
            });
        }

        self.balance -= amount;
        Ok(self.balance)
    }
}
// src/main.rs
//! Application entry point: Pattern matching on domain errors.

mod error;
mod bank;

use bank::Account;
use error::BankError;

fn process_transaction(acc: &mut Account, amount: f64) {
    match acc.withdraw(amount) {
        Ok(new_balance) => println!("Success! Remaining balance: ${new_balance:.2}"),
        Err(err) => match err {
            // Strong domain matching enables custom recovery paths!
            BankError::InsufficientFunds { requested, available } => {
                println!("[WARN] Failed: Requested ${requested:.2}, but only ${available:.2} exists.");
            }
            BankError::AccountLocked(id) => {
                println!("[ALERT] Security Incident: Account {id} attempted access while locked!");
            }
            BankError::NetworkTimeout => {
                println!("[RETRY] Retrying network transaction...");
            }
        },
    }
}

fn main() {
    let mut acc = Account::new("ACC-9042", 50.00);
    process_transaction(&mut acc, 120.00); // Will trigger InsufficientFunds
}

Level 3 Architectural Pillars

Domain Enums vs Generic Strings

Avoid returning raw `String` or `&str` errors. Enum errors force calling code to handle every possible failure case exhaustively during compilation.

CS Principle: Exhaustive Algebraic Data Types

The ? Operator & From

The ? operator automatically calls From::from() under the hood, seamlessly coercing lower-level errors (like IO or database errors) into your high-level domain error.

CS Principle: Implicit Monadic Error Coercion

The std::error::Error Trait

Implementing `Display` and `std::error::Error` allows third-party ecosystem tools, loggers, and tracing frameworks to print backtraces and inspect error sources seamlessly.

CS Principle: Standardized Diagnostic Protocols

Library vs Application Split

Libraries use structured enums (e.g., via `thiserror`). Applications use dynamic context handlers (e.g., `anyhow` or `eyre`) to wrap errors with contextual stack traces.

CS Principle: Contextual Stack Wrapping
