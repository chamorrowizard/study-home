// Cargo.toml requirement: [lib] proc-macro = true
// my_macro_crate/src/lib.rs

use proc_macro::TokenStream;
use quote::quote;
use syn::{parse_macro_input, DeriveInput};

/// Custom derive macro generating metadata inspecting trait implementation
#[proc_macro_derive(DescribeStruct)]
pub fn derive_describe_struct(input: TokenStream) -> TokenStream {
    // 1. Parse raw TokenStream into Abstract Syntax Tree (AST)
    let input = parse_macro_input!(input as DeriveInput);
    let struct_name = &input.ident;

    // 2. Build generated Rust code AST replacement
    let expanded = quote! {
        impl #struct_name {
            pub fn describe() -> &'static str {
                concat!("Struct Name: ", stringify!(#struct_name))
            }
        }
    };

    // 3. Return generated code converted back into compiler TokenStream
    TokenStream::from(expanded)
}
// my_macro_crate/src/lib.rs
use proc_macro::TokenStream;
use quote::quote;
use syn::{parse_macro_input, ItemFn};

/// Custom attribute macro that benchmarks function execution time automatically
#[proc_macro_attribute]
pub fn time_execution(_attr: TokenStream, item: TokenStream) -> TokenStream {
    let input_fn = parse_macro_input!(item as ItemFn);
    let fn_vis = &input_fn.vis;
    let fn_sig = &input_fn.sig;
    let fn_block = &input_fn.block;
    let fn_name = &fn_sig.ident;

    let expanded = quote! {
        #fn_vis #fn_sig {
            let start = std::time::Instant::now();
            let result = (|| #fn_block )();
            println!("[TIMER] Execution of {} took: {:?}", stringify!(#fn_name), start.elapsed());
            result
        }
    };

    TokenStream::from(expanded)
}
// main_crate/src/main.rs
// Consuming custom proc-macros created in the macro crate

use my_macro_crate::{DescribeStruct, time_execution};

#[derive(DescribeStruct)]
struct UserSession {
    user_id: u64,
    token: String,
}

#[time_execution]
fn compute_heavy_payload() -> u64 {
    let mut sum = 0;
    for i in 0..1_000_000 {
        sum += i;
    }
    sum
}

fn main() {
    // Generated via #[derive(DescribeStruct)]
    println!("{}", UserSession::describe()); // Prints: Struct Name: UserSession

    // Instrumented via #[time_execution]
    let result = compute_heavy_payload();
    println!("Computed Result: {}", result);
}

Level 8 Architectural Pillars

TokenStream Processing

Procedural macros receive raw token streams directly from the Rust compiler before semantic analysis, transforming code at the lexical token stage.

Core Mandate: Lexical Compile-Time Transformation

syn AST Parsing Engine

The syn crate converts raw tokens into strongly-typed Abstract Syntax Tree structures (DeriveInput, ItemFn, Expr) for safe programmatic inspection.

Core Mandate: Safe Type-Driven AST Traversal

quote! Templating & Synthesis

The quote crate provides quasi-quoting capability using variable interpolation (#var) to emit syntactically sound Rust code streams cleanly.

Core Mandate: Zero-Cost Safe Code Synthesizer

Hygiene & Span Diagnostics

Macros maintain token Span metadata. This ensures compile errors in macro-generated code point directly back to the exact line in the consumer's source code.

Core Mandate: Rich Compiler Error Span Reporting
