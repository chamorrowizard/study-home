// 1. Low-Level Manual Future Implementation
// Demonstrates how Tokio registers interest via Context and Waker callbacks.

use std::future::Future;
use std::pin::Pin;
use std::task::{Context, Poll};
use std::time::{Duration, Instant};

pub struct YieldDelay {
    when: Instant,
}

impl YieldDelay {
    pub fn new(duration: Duration) -> Self {
        Self { when: Instant::now() + duration }
    }
}

impl Future for YieldDelay {
    type Output = &'static str;

    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll {
        if Instant::now() >= self.when {
            Poll::Ready("Delay Complete")
        } else {
            // Register current thread's waker to re-poll when timer fires
            let waker = cx.waker().clone();
            let when = self.when;
            
            std::thread::spawn(move || {
                let now = Instant::now();
                if when > now {
                    std::thread::sleep(when - now);
                }
                waker.wake(); // Re-wake executor loop
            });

            Poll::Pending
        }
    }
}
// 2. Concurrent Actor Pattern using Tokio MPSC & Oneshot Channels
use tokio::sync::{mpsc, oneshot};

enum Command {
    GetStatus { resp: oneshot::Sender },
    Increment,
}

#[tokio::main]
async fn main() {
    let (tx, mut rx) = mpsc::channel::(32);

    // Spawn Actor task loop
    tokio::spawn(async move {
        let mut count = 0;
        while let Some(cmd) = rx.recv().await {
            match cmd {
                Command::Increment => count += 1,
                Command::GetStatus { resp } => {
                    let _ = resp.send(format!("Current Count: {}", count));
                }
            }
        }
    });

    // Client messaging
    tx.send(Command::Increment).await.unwrap();
    let (resp_tx, resp_rx) = oneshot::channel();
    tx.send(Command::GetStatus { resp: resp_tx }).await.unwrap();

    let status = resp_rx.await.unwrap();
    println!("Actor State: {}", status);
}
// 3. Multi-branch Cancellation Safety via tokio::select!
use tokio::time::{sleep, Duration};

async fn fetch_remote_data() -> &'static str {
    sleep(Duration::from_millis(150)).await;
    "Payload Data Received"
}

#[tokio::main]
async fn main() {
    let timeout = sleep(Duration::from_millis(100));
    tokio::pin!(timeout); // Pin to stack so it can be safely polled in select!

    tokio::select! {
        res = fetch_remote_data() => {
            println!("Success: {}", res);
        }
        _ = &mut timeout => {
            println!("Error: Remote request timed out!");
        }
    } // Uncompleted branch is safely dropped (cancellation safety guaranteed)
}

Level 6 Architectural Pillars

Cooperative Lazy Futures

Rust futures do no background computation automatically. They are state machines stepped explicitly by the runtime via poll() returning Poll::Pending or Poll::Ready.

Runtime Principle: Zero-Cost Lazy State Machines

Pin<P> & Unpin Guarantees

Async blocks compile down to self-referential structs holding internal pointers across .await yields. Pin ensures memory addresses remain static so internal pointers never invalidate.

Runtime Principle: Safe Self-Referential Pointer Invariance

Task Waker Notification Protocol

Avoids spinning CPU loops. When an I/O handle is unblocked, it calls waker.wake(), signalling Tokio's event loop to put the dormant task back on the active execution queue.

Runtime Principle: Event-Driven Reaction Engine

Cancellation Safety in select!

When one branch completes in tokio::select!, all other incomplete futures are instantly dropped. Ensuring code is cancellation-safe avoids dropping partially processed socket data.

Runtime Principle: RAII Scope Drops for Unfinished Tasks
