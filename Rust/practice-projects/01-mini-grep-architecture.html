use std::error::Error;
use std::fs;
use std::env;

/// Configuration parameters for the minigrep engine.
pub struct Config {
    pub query: String,
    pub file_path: String,
    pub ignore_case: bool,
}

impl Config {
    /// Parses command-line arguments using an iterator.
    /// Employs ownership transfer via iterator consumption instead of cloning slices.
    pub fn build(mut args: impl Iterator) -> Result {
        // Skip executable path
        args.next();

        let query = match args.next() {
            Some(arg) => arg,
            None => return Err("Missing query string parameter"),
        };

        let file_path = match args.next() {
            Some(arg) => arg,
            None => return Err("Missing file path parameter"),
        };

        // Environment variable takes priority if IGNORE_CASE is set
        let ignore_case = env::var("IGNORE_CASE").is_ok();

        Ok(Config {
            query,
            file_path,
            ignore_case,
        })
    }
}

/// Executes the file search using zero-cost iterator adapters.
pub fn run(config: Config) -> Result<(), Box> {
    let contents = fs::read_to_string(&config.file_path)?;

    let results = if config.ignore_case {
        search_case_insensitive(&config.query, &contents)
    } else {
        search(&config.query, &contents)
    };

    for line in results {
        println!("{line}");
    }

    Ok(())
}

/// Case-sensitive line filtering using lazy iterator chaining.
pub fn search<'a>(query: &str, contents: &'a str) -> Vec<&'a str> {
    contents
        .lines()
        .filter(|line| line.contains(query))
        .collect()
}

/// Case-insensitive search using string lower-casing.
pub fn search_case_insensitive<'a>(query: &str, contents: &'a str) -> Vec<&'a str> {
    let query = query.to_lowercase();
    contents
        .lines()
        .filter(|line| line.to_lowercase().contains(&query))
        .collect()
}

Core Engineering & CS Essentials

Theoretical Computer Science meets Systems Engineering

Lifetime & Zero-Copy Views

Memory Efficiency // O(1) Overhead

The return signature Vec<&'a str> uses lifetime annotations to bind returned line references directly to the underlying contents String buffer. Instead of allocating new strings for matching lines, minigrep returns pointer slices—achieving near-zero heap allocations during search operations.

Lazy Iterator Abstractions

Zero-Cost Abstraction // SIMD Optimization

contents.lines().filter(...) creates a stateless iterator chain. No memory is evaluated or collected until .collect() is invoked. Rust's LLVM backend unrolls these iterators into vectorized instructions that rival hand-written C loops in speed.

Deterministic Error Boundaries

Fault Tolerance // No Panics

By replacing panic! calls with Result<T, E> and the propagation operator (?), errors are converted into first-class values. The calling function retains total operational control over failure handling, preventing unexpected process terminates in library crates.

I/O Bottlenecks & Cache Locality

Disk I/O vs Memory Bandwidth

fs::read_to_string performs a single contiguous system call read. Loading the entire file into CPU L1/L2 cache-friendly sequential memory makes subsequent text searches limited purely by memory bandwidth, rather than repetitive kernel context switches.

System Evolution Roadmap

Scaling minigrep from local CLI to Distributed Search Engine

STAGE 1

Single Threaded CLI (Current State)

Loads file synchronously from local disk into memory. Processed sequentially on a single CPU core. Ideal for files up to a few hundred megabytes.

Max File Size: ~RAM Limit • Throughput: ~500 MB/s
STAGE 2

Multi-Threaded Parallel Search (ripgrep Model)

Introduce memory mapping (mmap) and work-stealing thread pools (rayon). Split multi-gigabyte files into discrete byte chunks across CPU cores without copying.

Concurrency: Multi-core Rayon • Throughput: ~10 GB/s
STAGE 3

Distributed Indexing Cluster (ElasticSearch / Meilisearch Model)

Transition from line-scanning to pre-indexed inverted indexes. Partition files across storage nodes using Consistent Hashing. Implement gRPC endpoints for query dispatching and scatter-gather response aggregation.

Scale: Petabytes • Latency: Sub-10ms queries
