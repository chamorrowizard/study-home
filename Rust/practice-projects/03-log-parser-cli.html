use std::collections::HashMap;

#[derive(Debug, PartialEq, Eq, Hash, Clone)]
pub enum LogLevel {
    Info,
    Warn,
    Error,
    Unknown(String),
}

impl LogLevel {
    pub fn from_str(s: &str) -> Self {
        match s.to_uppercase().as_str() {
            "INFO" => LogLevel::Info,
            "WARN" | "WARNING" => LogLevel::Warn,
            "ERROR" | "ERR" => LogLevel::Error,
            other => LogLevel::Unknown(other.to_string()),
        }
    }
}

#[derive(Debug)]
pub struct LogEntry {
    pub timestamp: String,
    pub level: LogLevel,
    pub message: String,
}

impl LogEntry {
    /// Parses a single log line in the format: [TIMESTAMP] [LEVEL] MESSAGE
    pub fn parse_line(line: &str) -> Option {
        let parts: Vec<&str> = line.splitn(3, ' ').collect();
        if parts.len() < 3 {
            return None;
        }

        let timestamp = parts[0].trim_matches(|c| c == '[' || c == ']').to_string();
        let level = LogLevel::from_str(parts[1].trim_matches(|c| c == '[' || c == ']'));
        let message = parts[2].to_string();

        Some(LogEntry { timestamp, level, message })
    }
}

pub struct LogAggregator {
    pub entries: Vec,
}

impl LogAggregator {
    pub fn from_raw_logs(logs: &str) -> Self {
        let entries = logs
            .lines()
            .filter_map(LogEntry::parse_line)
            .collect();
        LogAggregator { entries }
    }

    pub fn count_by_level(&self) -> HashMap {
        let mut counts = HashMap::new();
        for entry in &self.entries {
            *counts.entry(entry.level.clone()).or_insert(0) += 1;
        }
        counts
    }

    pub fn filter_by_level(&self, target_level: LogLevel) -> Vec<&LogEntry> {
        self.entries
            .iter()
            .filter(|e| e.level == target_level)
            .collect()
    }
}

fn main() {
    let raw_logs = "\
[2026-08-07 10:00:01] [INFO] Server started successfully
[2026-08-07 10:00:05] [WARN] High memory usage detected: 85%
[2026-08-07 10:00:12] [ERROR] Failed to connect to primary database
[2026-08-07 10:00:15] [INFO] Retrying database connection...";

    let aggregator = LogAggregator::from_raw_logs(raw_logs);
    println!("Total Valid Entries: {}", aggregator.entries.len());

    let errors = aggregator.filter_by_level(LogLevel::Error);
    println!("\nCritical Error Logs ({}):", errors.len());
    for err in errors {
        println!(" - [{}] {}", err.timestamp, err.message);
    }
}

Architecture Breakdown

1. Type-Safe Severity Enums

Captures known log levels explicitly while using a fallback variant LogLevel::Unknown(String) for unexpected strings.

2. Fail-Safe Line Parsing

Utilizes splitn(3, ' ') and returns Option<LogEntry> to safely discard corrupted lines without crashing execution.

3. Zero-Allocation Filtering

Uses filter_map() and borrowed reference iterators to stream log processing through pipeline stages cleanly.

4. Aggregator Pipeline

Groups occurrences into key-value counts using `HashMap`, facilitating rapid metrics generation for monitoring outputs.
