#[derive(Debug, Clone)]
pub enum OpCode {
    LoadConst(u8, i64), // LoadConst(Register, Value)
    Add(u8, u8, u8),    // Add(TargetReg, RegA, RegB)
    Store(String, u8),  // Store(VarName, Reg)
    Halt,
}

pub struct VirtualMachine {
    registers: [i64; 8],
    variables: std::collections::HashMap,
}

impl VirtualMachine {
    pub fn new() -> Self {
        Self {
            registers: [0; 8],
            variables: std::collections::HashMap::new(),
        }
    }

    pub fn execute(&mut self, bytecode: &[OpCode]) {
        let mut ip = 0; // Instruction Pointer

        while ip < bytecode.len() {
            match &bytecode[ip] {
                OpCode::LoadConst(reg, val) => {
                    self.registers[*reg as usize] = *val;
                }
                OpCode::Add(target, reg_a, reg_b) => {
                    let val_a = self.registers[*reg_a as usize];
                    let val_b = self.registers[*reg_b as usize];
                    self.registers[*target as usize] = val_a + val_b;
                }
                OpCode::Store(var_name, reg) => {
                    let val = self.registers[*reg as usize];
                    self.variables.insert(var_name.clone(), val);
                }
                OpCode::Halt => break,
            }
            ip += 1;
        }
    }

    pub fn get_var(&self, name: &str) -> Option<&i64> {
        self.variables.get(name)
    }
}

fn main() {
    // Compiled instruction stream corresponding to expression: "x = 15 + 25"
    let program = vec![
        OpCode::LoadConst(0, 15),       // R0 = 15
        OpCode::LoadConst(1, 25),       // R1 = 25
        OpCode::Add(2, 0, 1),           // R2 = R0 + R1
        OpCode::Store("x".to_string(), 2), // x = R2
        OpCode::Halt,
    ];

    let mut vm = VirtualMachine::new();
    vm.execute(&program);

    if let Some(val) = vm.get_var("x") {
        println!("Executed Expression. Output Variable x = {val}");
    }
}

Architecture Breakdown

1. Token Lexing & Lexical Analysis

Converts raw string source code into a clean, iterable sequence of tokens, stripping whitespace and identifying operators or numbers.

2. Recursive Descent Parsing (AST)

Parses structural token hierarchies into an Abstract Syntax Tree (AST) while preserving operator precedence rules.

3. OpCode Emission

Traverses the AST to produce linear byte code instructions for register loading, arithmetic operations, and variable storage.

4. Register Machine Execution Loop

Dispatches Instructions via a fast loop over fixed registers, keeping execution state compact and CPU cache-friendly.
