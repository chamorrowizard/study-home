// 1. Signature: Takes u64, returns an Option<u64>
fn find_next_square(sq: u64) -> Option<u64> {
    // 2. Cast & Sqrt
    let root_float = (sq as f64).sqrt();

    // 3. Integer Check
    if root_float.fract() == 0.0 {
        // 4. Cast back & calculate next root
        let current_root = root_float as u64;
        let next_root = current_root + 1;
        
        // 5. Return success result wrapped in Some()
        Some(next_root * next_root)
    } else {
        // 6. Return failure wrapped in None
        None      
    }
}
